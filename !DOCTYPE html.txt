    function setCartInfo() {
      const total = state.cart.reduce((sum, id) => {
        const p = PRODUCTS.find(x => x.id === id);
        return sum + (p ? p.price : 0);
      }, 0);
      qs('#cartCount').textContent = String(state.cart.length);
      qs('#cartTotal').textContent = PEN.format(total);
    }
    function addToCart(id) {
      state.cart.push(id);
      setCartInfo();
    }

    function productCard(p){
      const div = document.createElement('div');
      div.className = 'card';
      div.innerHTML = `
        <img src="${p.img}" alt="${p.name}" style="height:180px; object-fit:cover;"/>
        <div class="p">

          <div style="display:flex; align-items:center; justify-content:space-between; margin-top:8px;">
            <strong>${PEN.format(p.price)}</strong>
            <button class="btn" data-add="${p.id}">Añadir</button>
          </div>
        </div>`;
      div.addEventListener('click',(e)=>{
        const btn = e.target.closest('[data-add]');
        if(btn){ addToCart(p.id); }
      });
      return div;
    }

      SEASONS.forEach(s => { const o=document.createElement('option'); o.value=s; o.textContent=s; seasonSel.appendChild(o); });
    }
    function renderDesigner(){
      const grid = qs('#designerGrid');
      grid.innerHTML = '';
      const style = qs('#styleSelect').value;
      const color = qs('#colorSelect').value;
      const results = PRODUCTS.filter(p => (!style || p.style === style) && (!color || p.color === color));
      results.forEach(p => grid.appendChild(productCard(p)));
      return results;
    }

      const grid = qs('#ugcGrid'); grid.innerHTML='';
